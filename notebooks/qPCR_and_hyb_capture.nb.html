test_dt <- merge_dt |>
  mutate(
    seq_detect = case_when(
      RPKMF <= 0 ~ FALSE,
      TRUE ~ TRUE
      ),
    pcr_count = rowSums(!is.na(across(c(Ct_1a, Ct_2a, Ct_2b))))
  ) |>
  group_by(patient_number) |>
  summarize(
    n = n(),
    n_seq_detect = sum(seq_detect, na.rm = TRUE),
    pcr_total = sum(pcr_count)
    ) |>
  mutate(
    pat_seq_detect = case_when(
      n_seq_detect <= 0 ~ FALSE,
      TRUE ~ TRUE
      ),
    pat_pcr_detect = case_when(
      pcr_total <= 0 ~ FALSE,
      TRUE ~ TRUE
      )
  )
